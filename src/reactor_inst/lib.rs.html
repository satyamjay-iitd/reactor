15// Create a Resource that captures information about the entity for which telemetry is recorded.
16fn resource() -> Resource {
17    Resource::builder()
18        .with_service_name(env!("CARGO_PKG_NAME"))
19        .with_schema_url(
20            [
21                KeyValue::new(SERVICE_VERSION, env!("CARGO_PKG_VERSION")),
22                KeyValue::new(DEPLOYMENT_ENVIRONMENT_NAME, "develop"),
23            ],
24            SCHEMA_URL,
25        )
26        .build()
27}
28pub struct OtelGuard {
29    tracer_provider: SdkTracerProvider,
30    meter_provider: SdkMeterProvider,

33impl Drop for OtelGuard {
34    fn drop(&mut self) {
35        if let Err(err) = self.tracer_provider.shutdown() {
36            eprintln!("{err:?}");
37        }
38        if let Err(err) = self.meter_provider.shutdown() {
39            eprintln!("{err:?}");
40        }
41    }
42}

44// Initialize tracing-subscriber and return OtelGuard for opentelemetry-related termination processing
45pub fn init_tracing() -> OtelGuard {
46    let tracer_provider = init_tracer_provider();
47    let meter_provider = init_meter_provider();
48
49    let tracer = tracer_provider.tracer("reactor");
50
51    tracing_subscriber::registry()
52        // The global level filter prevents the exporter network stack
53        // from reentering the globally installed OpenTelemetryLayer with
54        // its own spans while exporting, as the libraries should not use
55        // tracing levels below DEBUG. If the OpenTelemetry layer needs to
56        // trace spans and events with higher verbosity levels, consider using
57        // per-layer filtering to target the telemetry layer specifically,
58        // e.g. by target matching.
59        .with(tracing_subscriber::filter::LevelFilter::from_level(
60            Level::INFO,
61        ))
62        .with(tracing_subscriber::fmt::layer())
63        // .with(MetricsLayer::new(meter_provider.clone()))
64        .with(OpenTelemetryLayer::new(tracer))
65        .init();
66
67    OtelGuard {
68        tracer_provider,
69        meter_provider,
70    }
71}

73// Construct TracerProvider for OpenTelemetryLayer
74fn init_tracer_provider() -> SdkTracerProvider {
75    let exporter = opentelemetry_otlp::SpanExporter::builder()
76        .with_tonic()
77        .build()
78        .unwrap();
79
80    SdkTracerProvider::builder()
81        // Customize sampling strategy
82        // .with_sampler(Sampler::ParentBased(Box::new(Sampler::TraceIdRatioBased(
83        //     1.0,
84        // ))))
85        .with_sampler(Sampler::AlwaysOn)
86        // If export trace to AWS X-Ray, you can use XrayIdGenerator
87        .with_id_generator(RandomIdGenerator::default())
88        .with_resource(resource())
89        .with_batch_exporter(exporter)
90        .build()
91}

93// Construct MeterProvider for MetricsLayer
94fn init_meter_provider() -> SdkMeterProvider {
95    let exporter = opentelemetry_otlp::MetricExporter::builder()
96        .with_tonic()
97        .with_temporality(opentelemetry_sdk::metrics::Temporality::default())
98        .build()
99        .unwrap();
100
101    let reader = PeriodicReader::builder(exporter)
102        .with_interval(std::time::Duration::from_secs(30))
103        .build();
104
105    // For debugging in development
106    let stdout_reader =
107        PeriodicReader::builder(opentelemetry_stdout::MetricExporter::default()).build();
108
109    let meter_provider = MeterProviderBuilder::default()
110        .with_resource(resource())
111        .with_reader(reader)
112        .with_reader(stdout_reader)
113        .build();
114
115    global::set_meter_provider(meter_provider.clone());
116
117    meter_provider
118}
