35/// Represents the routing target of a message in the system.
36///
37/// `RouteTo` is used to specify where a message should be sent. It supports
38/// multiple routing strategies:
39///
40/// - `Blackhole`: Drop the message silently without sending it.
41/// - `Reply`: Route the message back to the original sender (used in request-reply patterns).
42/// - `Single`: Route the message to a single actor by address.
43/// - `Multiple`: Route the message to multiple addresses of actors.
44///
45/// The `'a` lifetime allows `RouteTo` to borrow from external string data
46/// when possible, avoiding allocations when routing with references.
47pub enum RouteTo<'a> {
48    /// Drop the message; it will not be sent to any actor.
49    Blackhole,
50    /// Send the message back to the sender.
51    Reply,
52    /// Route to a single actor by address (either borrowed or owned).
53    Single(Cow<'a, str>),
54    /// Route to multiple actors by addresses (either borrowed or owned).
55    Multiple(Cow<'a, [String]>),
56}
57
58/// Converts a borrowed `&str` into a `RouteTo::Single` with borrowed data.
59/// Usage:
60///      RouteTo::from("addr".to_string());
61impl From<String> for RouteTo<'_> {
62    fn from(value: String) -> Self {
63        RouteTo::Single(Cow::Owned(value))
64    }
65}
66
67/// Converts a borrowed `&str` into a `RouteTo::Single` with borrowed data.
68/// Usage:
69///      RouteTo::from(&self.other_addr.as_str());    // other_addr: String
70impl<'a> From<&'a str> for RouteTo<'a> {
71    fn from(value: &'a str) -> Self {
72        RouteTo::Single(Cow::Borrowed(value))
73    }
74}
75
76/// Converts a `Vec<String>` into a `RouteTo::Multiple` with owned address list.
77/// Usage:
78///      let x = vec!["addr1".to_string];
79///      RouteTo::from(x);
80impl From<Vec<String>> for RouteTo<'_> {
81    fn from(value: Vec<String>) -> Self {
82        RouteTo::Multiple(Cow::Owned(value))
83    }
84}
85
86/// Converts a borrowed slice of `String`s into a `RouteTo::Multiple`.
87/// Usage:
88///      RouteTo::from(self.addrs.as_slice());  // addrs: Vec<String>
89impl<'a> From<&'a [String]> for RouteTo<'a> {
90    fn from(value: &'a [String]) -> Self {
91        RouteTo::Multiple(Cow::Borrowed(value))
92    }
93}

101/// Represents the action to take after receiving and decoding a message from a channel.
102///
103/// This enum is used by the message receiver to determine
104/// how to handle an incoming message based on its content and the current channel state.
105///
106/// # Variants
107///
108/// - `PASS`:
109///   Forward the message to the next stage in the pipeline (e.g., processor).
110///
111/// - `PANIC`:
112///   Indicates a critical error. Triggers a panic, typically used to fail fast on invalid or unexpected input.
113//
114/// - `DROP`:
115///   Silently discard the message without processing or forwarding.
116///
117/// - `SYNC(u16)`:
118///   Perform a synchronization operation with the provided sync identifier (e.g., a sync round or epoch number).
119///
120/// - `CLOSE`:
121///   Signals that the channel should be closed and no more messages will be received from it.
122#[derive(Debug)]
123pub enum ChannelAction {
124    PASS,
125    PANIC,
126    DROP,
127    SYNC(u16),
128    CLOSE,
129}

131#[derive(Debug)]
132enum R2PMsg<T> {
133    Msg(T, &'static str),
134    Exit,
135    #[allow(dead_code)]
136    AddPrio(mpsc::Receiver<R2PMsg<T>>),
137    #[allow(dead_code)]
138    RemoveLowPrio,
139}
140
141impl<T: Clone> Clone for R2PMsg<T> {
142    fn clone(&self) -> Self {
143        match self {
144            R2PMsg::Msg(m, origin) => R2PMsg::Msg(m.clone(), origin),
145            _ => panic!("Shouldn't Clone this"),
146        }
147    }
148}
149impl<T: PartialEq> PartialEq for R2PMsg<T> {
150    fn eq(&self, other: &Self) -> bool {
151        match (self, other) {
152            (R2PMsg::AddPrio(_), _) => panic!("Can't Compare"),
153            (R2PMsg::Msg(m1, _), R2PMsg::Msg(m2, _)) => m1.eq(m2),
154            (R2PMsg::Exit, R2PMsg::Exit) => true,
155            _ => false,
156        }
157    }
158}
159
160impl<T: HasPriority> HasPriority for R2PMsg<T> {
161    fn priority(&self) -> usize {
162        match self {
163            R2PMsg::Msg(t, _) => t.priority(),
164            R2PMsg::Exit => MAX_PRIO,
165            R2PMsg::AddPrio(_) => MAX_PRIO,
166            R2PMsg::RemoveLowPrio => MAX_PRIO,
167        }
168    }
169}

171/// The `ActorRecv` trait defines what action to take based on incoming message.
172///
173/// It defines a single asynchronous method, [`ActorRecv::after_recv`], that is called after a message is received.
174///
175/// # Type Parameters
176/// - `IMsg`: The type of the message this actor receives. It must implement the [`Msg`] trait.
177///
178pub trait ActorRecv: Send + 'static {
179    /// The input message type that this actor receives.
180    type IMsg: Msg;
181    /// Called after the actor receives a message.
182    ///
183    /// This asynchronous method is invoked with:
184    /// - `worker_id`: A reference to the address of the sending actor.
185    /// - `input`: A reference to the message that was received.
186    ///
187    /// It returns [`ChannelAction`] that determines how the actor should proceed.
188    fn after_recv(
189        &mut self,
190        worker_id: &str,
191        input: &Self::IMsg,
192    ) -> impl std::future::Future<Output = ChannelAction> + Send;
193}

205/// The `ActorProcess` trait defines the processing logic for an actor that transforms
206/// input messages into one or more output messages.
207///
208/// # Example
209/// ```ignore
210/// struct Incrementer;
211///
212/// impl ActorProcess for Incrementer {
213///     type IMsg = i32;
214///     type OMsg = i32;
215///
216///     fn process(&mut self, input: i32) -> Vec<i32> {
217///         vec![input + 1]
218///     }
219/// }
220///
221pub trait ActorProcess: Send + 'static {
222    /// The type of messages this actor accepts as input.
223    type IMsg: Msg;
224
225    /// The type of messages this actor produces as output.
226    type OMsg: Msg;
227
228    /// Processes an input message and returns a list of output messages.
229    ///
230    /// # Arguments
231    ///
232    /// * `input` - The input message to be processed.
233    ///
234    /// # Returns
235    ///
236    /// A vector of output messages of type [`Self::OMsg`].
237    fn process(&mut self, input: Self::IMsg) -> Vec<Self::OMsg>;
238}

240/// The `ActorSend` trait defines how an actor determines the recipients of a message
241/// before it is sent.
242///
243/// # Example
244/// ```ignore
245/// struct Broadcaster {
246///     peers: Vec<ActorAddrRef>,
247/// }
248///
249/// impl ActorSend for Broadcaster {
250///     type OMsg = MyMessage;
251///
252///     async fn before_send<'a>(
253///         &'a mut self,
254///         _output: &Self::OMsg,
255///     ) -> &'a Vec<ActorAddrRef> {
256///         &self.peers
257///     }
258/// }
259///
260pub trait ActorSend: Send + 'static {
261    /// The type of output messages that this actor sends.
262    type OMsg: Msg;
263
264    /// Called before an output message is sent.
265    ///
266    /// This asynchronous method returns the list of recipients that the message should be sent to.
267    ///
268    /// # Arguments
269    ///
270    /// * `output` - A reference to the message that is about to be sent.
271    ///
272    /// # Returns
273    ///
274    /// a list of [`ActorAddrRef`] indicating the recipient actors.
275    fn before_send<'a>(
276        &'a mut self,
277        output: &Self::OMsg,
278    ) -> impl std::future::Future<Output = RouteTo<'a>> + Send;
279}

291/// The `Behaviour` struct encapsulates the complete behavior of an actor,
292/// including how it receives messages, processes them, sends output,
293/// and optionally generates new messages.
294///
295/// # Type Parameters
296///
297/// - `R`: The type implementing the receiving behavior (must implement [`ActorRecv`]).
298/// - `P`: The type implementing the processing behavior (must implement [`ActorProcess`]).
299/// - `S`: The type implementing the sending behavior (must implement [`ActorSend`]).
300/// - `M`: The type of message generated internally (e.g., from generators).
301///
302/// # Fields
303///
304/// - `recv`: Optional receiver logic implementing `ActorRecv`.
305/// - `proc`: The core processing logic implementing `ActorProcess`.
306/// - `send`: Optional sender logic implementing `ActorSend`.
307/// - `generators`: A list of internal message generators, producing messages of type `M`.
308///
309pub struct Behaviour<R, P, S, M: 'static, MCD> {
310    recv: Option<R>,
311    proc: P,
312    send: Option<S>,
313    generators: Vec<Box<dyn Iterator<Item = M> + Send>>,
314    num_prios: u8,
315    master_codec: MCD,
316    sub_decoders: Option<SubDecoderStore<M>>,
317    receiver_should_adapt: bool,
318}

468impl<R, P, S, IM, OM, MCD> Behaviour<R, P, S, IM, MCD>
469where
470    IM: Msg,
471    OM: Msg,
472    R: ActorRecv<IMsg = IM>,
473    P: ActorProcess<IMsg = IM, OMsg = OM>,
474    S: ActorSend<OMsg = OM>,
475    MCD: Encoder<OM> + Decoder<Item = IM, Error = std::io::Error> + Send + Sync + Clone + 'static,
476{
477    pub async fn run(mut self, ctx: RuntimeCtx) -> Result<(), ActorError> {
478        // let my_addr = ctx.addr.to_string();
479        let (p2s_tx, p2s_rx) = mpsc::unbounded_channel::<(OM, &'static str)>();
480        let (r2p_tx, mut r2p_rx) = reactor_channel::<R2PMsg<IM>>(self.num_prios, CHANNEL_SIZE);
481
482        let (controller_rx, controller_tx) = ctx.node_comm.split();
483
484        let reciever = self.take_recv();
485        let sender = self.take_send();
486        let mut generators = self.take_generators();
487        let mut processor = self.proc;
488
489        let gen_handles: Vec<tokio::task::JoinHandle<_>> = generators
490            .drain(..)
491            .map(|gene| {
492                let tx = r2p_tx.clone();
493                tokio::spawn(generator(gene, tx))
494            })
495            .collect();
496
497        let rx_handle = tokio::spawn(rx(
498            ctx.addr,
499            reciever,
500            r2p_tx,
501            self.master_codec.clone(),
502            self.sub_decoders,
503            controller_rx,
504        ));
505
506        let _addr = ctx.addr;
507        let proc_handle: JoinHandle<Result<(), ActorError>> =
508            tokio::task::spawn_blocking(move || -> Result<(), ActorError> {
509                tracing::info!("[ACTOR][{}] Processor Started", _addr);
510                loop {
511                    match r2p_rx.recv() {
512                        Some(R2PMsg::Msg(m, origin)) => {
513                            let processed = processor.process(m);
514                            for o in processed {
515                                p2s_tx.send((o, origin)).map_err(|_| ActorError::P2SErr)?;
516                            }
517                        }
518                        Some(R2PMsg::AddPrio(new_rx)) => {
519                            r2p_rx = r2p_rx.add_prio(new_rx);
520                        }
521                        Some(R2PMsg::RemoveLowPrio) => {
522                            r2p_rx = r2p_rx.remove_prio();
523                        }
524                        Some(R2PMsg::Exit) => {
525                            break;
526                        }
527                        None => {
528                            break;
529                        }
530                    }
531                }
532                tracing::info!("[ACTOR][{}] Processor Ended", _addr);
533                Ok(())
534            });
535        let tx_handle = tokio::spawn(tx(
536            ctx.addr,
537            sender,
538            self.receiver_should_adapt,
539            p2s_rx,
540            controller_tx,
541            self.master_codec,
542        ));
543        rx_handle.await??;
544        proc_handle.await??;
545        tx_handle.await?;
546        join_all(gen_handles).await;
547        Ok(())
548    }
549}
